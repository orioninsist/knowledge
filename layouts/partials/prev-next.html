{{- $current := . -}} {{- $pages := $current.CurrentSection.RegularPages.ByTitle -}} {{- $count := len $pages -}} {{- if gt $count 1 -}} {{- $currentIndex := -1 -}} {{- range $index, $page := $pages -}} {{- if eq $page.RelPermalink $current.RelPermalink -}} {{- $currentIndex = $index -}} {{- end -}} {{- end -}} {{- if ge $currentIndex 0 -}} {{- $previous := false -}} {{- $next := false -}} {{- if gt $currentIndex 0 -}} {{- $previous = index $pages (sub $currentIndex 1) -}} {{- end -}} {{- if lt $currentIndex (sub $count 1) -}} {{- $next = index $pages (add $currentIndex 1) -}} {{- end -}}
{{- with $previous -}} ← Previous {{ .Title }} {{- end -}}
{{- with $next -}} Next → {{ .Title }} {{- end -}}
{{- end -}} {{- end -}}
